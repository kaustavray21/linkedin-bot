# --- Discovery: throttles ----------------------------------------------------
# Requests per second is THE safety knob. Worker count follows from rate x
# latency rather than being chosen independently: at these rates 3 workers is
# the knee — 6 and 12 measured exactly as fast and only sent more requests
# after a block. If you start getting blocked, lower the rate; do not raise
# concurrency expecting it to help.
DISCOVERY_REQUESTS_PER_SECOND=2.0
DISCOVERY_CONCURRENCY=3
DISCOVERY_CONCURRENCY_MAX=6         # ceiling for the adaptive ramp
DISCOVERY_ADAPTIVE=true
DISCOVERY_RAMP_AFTER=5              # consecutive successes per ramp step

# Burst of 1 on purpose: a token bucket that starts full fires `rate` requests
# simultaneously on its first tick. Measured against an 8/s cap — 10.72/s
# observed with a full bucket, 7.93/s with a burst of 1.
DISCOVERY_TOKEN_BURST=1.0

# Optional extra floor between requests. 0 means the token bucket alone paces;
# kept as an escape hatch back to serial-ish behaviour with no code change.
DISCOVERY_MIN_INTERVAL_SECONDS=0.0
DISCOVERY_JITTER_SECONDS=0.0

# A backstop, not a pacer — the token bucket does the pacing.
DISCOVERY_DAILY_FETCH_CAP=300
DISCOVERY_CIRCUIT_THRESHOLD=3
DISCOVERY_CIRCUIT_COOLDOWN_HOURS=24
DISCOVERY_FETCH_TIMEOUT=20

# --- Discovery: retention ----------------------------------------------------
# expires_at is computed at INSERT, so changing this only affects new rows.
# Existing rows need a backfill (see migration c93a1f60d4e7).
DISCOVERY_RETENTION_DAYS=30
